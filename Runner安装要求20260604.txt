  安装步骤                                                                                                                                                  
                                                                                                                                                            
  # 1. 创建目录                                                                                                                                             
  mkdir -p ~/actions-runner && cd ~/actions-runner                                                                                                          
                                                                                                                                                            
  # 2. 下载 runner（x64 Linux）                                                                                                                             
  curl -fL https://github.com/actions/runner/releases/download/v2.323.0/actions-runner-linux-x64-2.323.0.tar.gz \                                           
    -o actions-runner.tar.gz                                                                                                                                
                                                                                                                                                            
  # 3. 解压                                                                                                                                                 
  tar -xzf actions-runner.tar.gz && rm actions-runner.tar.gz                                                                                                
                                                                                                                                                            
  # 4. 注册（--unattended 全默认，无需交互）                                                                                                                
  ./config.sh \                                                                                                                                             
    --url https://github.com/caoyb888/zhilian \                                                                                                             
    --token ANAZ3FFNSFP4UQOMWFYXZXLKEFBJU \                                                                                                                 
    --name intranet-runner \                                                                                                                                
    --labels "self-hosted,linux,intranet" \                                                                                                                 
    --work _work \                                                                                                                                          
    --unattended                                                                                                                                            
                                                                                                                                                            
  # 5. 安装为 systemd 服务（开机自启）                                                                                                                      
  sudo ./svc.sh install                                                                                                                                     
  sudo ./svc.sh start                                                                                                                                       
                                                                               
  # 6. 验证运行状态
  sudo ./svc.sh status
